cmake_minimum_required(VERSION 3.20)
project(HugeBoards LANGUAGES CXX)
if(NOT MSVC)
  message(FATAL_ERROR "This check requires MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(FSX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(GLOB_RECURSE ENGINE_SOURCES "${FSX_ROOT}/src/*.cpp")
list(FILTER ENGINE_SOURCES EXCLUDE REGEX "/(ffishjs|ffishdll|pyffish)\\.cpp$")
foreach(width 256 512)
  foreach(popcnt table hardware)
    set(target "bitboard-${width}-${popcnt}")
    add_executable(${target} "${FSX_ROOT}/tests/multiword-bitboard.cpp")
    target_compile_definitions(${target} PRIVATE BOARD_BITS=${width})
    target_compile_options(${target} PRIVATE /UNDEBUG)
    if(popcnt STREQUAL "hardware")
      target_compile_definitions(${target} PRIVATE USE_POPCNT)
    endif()
  endforeach()
  add_executable(engine-${width} ${ENGINE_SOURCES})
  target_compile_definitions(engine-${width} PRIVATE BOARD_BITS=${width} NNUE_EMBEDDING_OFF ALLVARS)
endforeach()
